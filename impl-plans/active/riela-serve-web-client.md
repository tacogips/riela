## 1. Objective and boundaries

Make bare `riela serve` serve the dashboard SPA whenever `RielaWebAssetLocator` can find
built web assets (including from a Homebrew-installed, symlinked binary), keep serving the
API either way, keep `--web-root` as an override with its existing validation, report
API-only mode clearly in the ready output, and ship/install the assets in the CLI archive and
formula. The design decisions are final; do not re-open them.

### Applicable prior knowledge

Knowledge-base recall for `web-asset-locator` returned no entries. Repository-derived facts
that stand in for it (verified in the analysis step):

- `Bundle.main.executableURL` returns the **symlink** path when a CLI is invoked through a
  symlink (`/opt/homebrew/bin/riela -> ../Cellar/...`); `Bundle.main.resourceURL` is the
  symlink's directory. `resolvingSymlinksInPath()` gives the real binary. (F3)
- `Formula/riela.rb` is generated by the heredoc in `scripts/render-homebrew-formula.sh`;
  editing only one of them causes drift at the next release. (F8)
- `linux-release.yml` runs `scripts/build-homebrew-release.sh` inside the official `swift`
  docker image, which has no `bun`. (F10)
- A stale `.build/*/riela` or stale `web/dist` silently produces misleading evidence — always
  rebuild before curling (verificationHint).
- `RielaStaticAssetResolver` had a `GET //` force-unwrap crash fixed earlier; its
  path handling is reused verbatim and must not be edited.

### 2.1 `Sources/RielaServer/RielaLocalHTTPServer.swift` — `RielaWebAssetLocator` (T1)

Replace the existing enum (lines ~347–368) with:

```swift
public enum RielaWebAssetLocator {
  /// Candidates tried relative to the *resolved* executable's directory, in order.
  /// `../Resources/Web` is the RielaApp.app / DMG layout; `../share/riela/web` is the
  /// Homebrew formula / CLI archive layout (`pkgshare`).
  public static let executableRelativeCandidates = ["../Resources/Web", "../share/riela/web"]

  public static func locate(
    bundle: Bundle? = .main,
    executableURL: URL? = Bundle.main.executableURL,
    currentDirectoryURL: URL = URL(fileURLWithPath: FileManager.default.currentDirectoryPath, isDirectory: true)
  ) -> URL? {
    candidates(bundle: bundle, executableURL: executableURL, currentDirectoryURL: currentDirectoryURL)
      .first { FileManager.default.fileExists(atPath: $0.appendingPathComponent("index.html").path) }
  }

  public static func candidates(
    bundle: Bundle? = .main,
    executableURL: URL? = Bundle.main.executableURL,
    currentDirectoryURL: URL = URL(fileURLWithPath: FileManager.default.currentDirectoryPath, isDirectory: true)
  ) -> [URL] {
    var candidates: [URL] = []
    if let resourceURL = bundle?.resourceURL {
      candidates.append(resourceURL.appendingPathComponent("Web", isDirectory: true))
    }
    if let executableURL {
      let executableDirectory = executableURL
        .resolvingSymlinksInPath()
        .standardizedFileURL
        .deletingLastPathComponent()
      for relative in executableRelativeCandidates {
        candidates.append(
          executableDirectory.appendingPathComponent(relative, isDirectory: true).standardizedFileURL
        )
      }
    }
    candidates.append(currentDirectoryURL.appendingPathComponent("web/dist", isDirectory: true))
    return candidates
  }
}
```

Rules: keep it outside the `#if canImport(Network)` block (Linux build); existing callers
(`Sources/RielaApp/EntryPoint+WebServer.swift:9` `RielaWebAssetLocator.locate()`) compile
unchanged. Order 1→4 must be exactly: bundle `Web`, `../Resources/Web`, `../share/riela/web`,
`cwd/web/dist`.

enum ServeWebRootSource: String, Sendable {
  case explicit = "--web-root"
  case located = "auto"
}

struct ServeWebRootResolution: Equatable, Sendable {
  var root: URL?
  var source: ServeWebRootSource?
  var diagnostic: String?

  static let missingAssetsDiagnostic =
    "missing; serving the API only. Build the dashboard with 'cd web && bun run build' or pass --web-root <dir>"
  static let unavailable = ServeWebRootResolution(root: nil, source: nil, diagnostic: missingAssetsDiagnostic)
  static func rejected(_ url: URL) -> ServeWebRootResolution {
    .init(root: nil, source: nil,
          diagnostic: "rejected \(url.path); serving the API only. Build the dashboard with 'cd web && bun run build' or pass --web-root <dir>")
  }
}

func resolvedServeWebRoot(
  parsed: ParsedParityOptions,
  locateDefault: () -> URL? = { RielaWebAssetLocator.locate() }
) throws -> ServeWebRootResolution {
  if let raw = parsed.webRoot?.trimmingCharacters(in: .whitespacesAndNewlines), !raw.isEmpty {
    return ServeWebRootResolution(root: try validatedServeWebRoot(raw), source: .explicit, diagnostic: nil)   // D1
  }
  guard let located = locateDefault() else { return .unavailable }                                          // D2
  guard let root = try? validatedServeWebRoot(located.path) else { return .rejected(located) }               // D3
  return ServeWebRootResolution(root: root, source: .located, diagnostic: nil)
}

/// The existing body of resolvedServeWebRoot from `let root = URL(fileURLWithPath: raw…)` to
/// `return root`, moved verbatim; error text unchanged:
/// CLIUsageError("--web-root requires a readable directory containing index.html")
private func validatedServeWebRoot(_ raw: String) throws -> URL
